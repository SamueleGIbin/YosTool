        const origOpen = XMLHttpRequest.prototype.open;
        const origSend = XMLHttpRequest.prototype.send;
        XMLHttpRequest.prototype.open = function (method, url) {
            this._tntUrl = url;
            return origOpen.apply(this, arguments);
        };
        XMLHttpRequest.prototype.send = function (body) {
            this.addEventListener("load", function () {

            });
            return origSend.apply(this, arguments);

    function sembraListaMovimenti(dati) {
        return Array.isArray(dati) && dati.length > 0 && dati[0] && typeof dati[0] === "object"

    function elaboreRispostaMovimenti(movimenti) {
        movimenti.forEach((m) => {
            if (!m || !Array.isArray(m.units)) return;
            m.units.forEach((u) => {
                if (!u || !u.unitId) return;
                cacheMovimentiPerUnit.set(String(u.unitId), {
                    origin: m.origin,
                    destination: m.destination,
                    scheduleArrival: m.scheduleArrival,
                    estimatedArrival: m.estimatedArrival,
                    actualArrival: m.actualArrival,
                    movementName: m.movementName,
                    tripTourDescription: m.trip ? m.trip.tripTourDescription : "",
                    vistoIl: Date.now()
                });
            });
        });
    }

    const FINESTRA_CONSERVAZIONE_CACHE_MOVIMENTI_MS = 16 * 60 * 60 * 1000;
    function pulisciRegistriSenzaBaia() {
        const ora = Date.now();
        cacheMovimentiPerUnit.forEach((mov, unitId) => {
            const arrivoIso = mov.actualArrival || mov.scheduleArrival || mov.estimatedArrival;
            const riferimento = arrivoIso ? new Date(arrivoIso).getTime() : mov.vistoIl;
            if (!riferimento || isNaN(riferimento) || (ora - riferimento) > FINESTRA_CONSERVAZIONE_CACHE_MOVIMENTI_MS) {
                cacheMovimentiPerUnit.delete(unitId);
            }
        });
        unitGiaScaricati.forEach((vistoIl, unitId) => {
            if (!vistoIl || (ora - vistoIl) > FINESTRA_CONSERVAZIONE_CACHE_MOVIMENTI_MS) unitGiaScaricati.delete(unitId);
        });
        const arrivi = caricaArriviBaia();
        let modificato = false;
        Object.keys(arrivi).forEach((doorNumber) => {
            const voce = arrivi[doorNumber];
            if (!voce || !voce.unitId || String(voce.unitId).indexOf("TRIP_") !== 0) return;
            const riferimento = voce.orario ? new Date(voce.orario).getTime() : null;
            if (!riferimento || isNaN(riferimento) || (ora - riferimento) > FINESTRA_CONSERVAZIONE_CACHE_MOVIMENTI_MS) {
                delete arrivi[doorNumber];
                modificato = true;
            }
        });
        if (modificato) salvaArriviBaia(arrivi);
    }

    function calcolaCutoffDaCacheMovimenti(doorNumber) {
        const arrivi = caricaArriviBaia();
        const voce = arrivi[doorNumber];
        if (!voce || !voce.unitId) return null;
        const mov = cacheMovimentiPerUnit.get(String(voce.unitId));
        if (!mov || !mov.origin) return null;
        const arrivoIso = mov.actualArrival || mov.scheduleArrival || mov.estimatedArrival;
        if (!arrivoIso) return null;
        const arrivo = new Date(arrivoIso);
        if (isNaN(arrivo.getTime())) return null;
        let arrivoPortaReale = null;
        if (voce.orario) {
            const d = new Date(voce.orario);
            if (!isNaN(d.getTime())) arrivoPortaReale = d;
        }
        const risultato = calcolaCutoffPerCodice(mov.origin, arrivo, arrivoPortaReale);
        if (!risultato) return null;
        return { cutoff: risultato, codice: mov.origin };
    }

        const n = parseInt(door, 10);
        if (isNaN(n)) return null;
        if (n >= 101 && n <= 107) return "Lato 100";

        if (n >= 601 && n <= 609) return "Lato 600";
        if (n >= 501 && n <= 509) return "Lato 500";
        return null;

    function avviaSessioneScarico(doorNumber, unitId, orarioIso) {
        const lato = latoBaia(doorNumber);
        if (!lato) return;
        const attuale = sessioniScaricoInCorso.get(doorNumber);
        if (attuale && attuale.unitId === String(unitId)) return;
        if (attuale) chiudiSessioneScarico(doorNumber, new Date().toISOString());
        sessioniScaricoInCorso.set(doorNumber, { unitId: String(unitId), lato, inizioIso: orarioIso });
    }

    function chiudiSessioneScarico(doorNumber, fineIso) {
        const attuale = sessioniScaricoInCorso.get(doorNumber);
        if (!attuale) return;
        sessioniScaricoInCorso.delete(doorNumber);
        const inizio = new Date(attuale.inizioIso).getTime();
        const fine = new Date(fineIso).getTime();
        if (isNaN(inizio) || isNaN(fine) || fine <= inizio) return;
        const durataMinuti = Math.round((fine - inizio) / 60000);
        if (durataMinuti <= 0 || durataMinuti > 12 * 60) return;
        const sessioni = caricaSessioniScarico();
        sessioni.push({ lato: attuale.lato, doorNumber, unitId: attuale.unitId, inizioIso: attuale.inizioIso, fineIso, durataMinuti });
        salvaSessioniScarico(sessioni);
        pianificaSincronizzazione(5000);
    }

            const esistente = arrivi[doorNumber];
            const cambiaUnit = !esistente || esistente.unitId !== unitId;
            if (cambiaUnit) {
                const sessionePrecedente = sessioniScaricoInCorso.get(doorNumber);
                if (sessionePrecedente && sessionePrecedente.unitId !== String(unitId)) chiudiSessioneScarico(doorNumber, new Date().toISOString());

            const azionePrecedente = esistente && esistente.action;

                && azionePrecedente !== unit.currentAction;
            if (cambiaUnit || transitaAdAzioneTracciata) {
                let orario;
                if (unit.currentAction === "START_UNLOADING" && (unit.updatedDateTimestamp || unit.updatedDate)) {
                    orario = unit.updatedDateTimestamp || unit.updatedDate;
                } else if (unit.currentAction === "LOCATION_ASSIGNED" && (unit.updatedDateTimestamp || unit.updatedDate)) {
                    orario = unit.updatedDateTimestamp || unit.updatedDate;
                } else {
                    orario = new Date().toISOString();
                }
                const orarioCambiato = !esistente || esistente.orario !== orario;
                if (cambiaUnit || orarioCambiato) {
                    cacheCutoffPerBaia.delete(doorNumber);
                    unitVerificatiCutoff.delete(String(unitId));
                }
                arrivi[doorNumber] = { unitId: unitId, orario: orario, action: unit.currentAction };
                modificato = true;

                if (cambiaUnit) registraInStoricoReport(doorNumber, unitId, orario);
                if (cambiaUnit && categoriaBaia(doorNumber) === "Scarico") avviaSessioneScarico(doorNumber, unitId, orario);

                arrivi[doorNumber] = { unitId: unitId, orario: esistente.orario, action: unit.currentAction };
                modificato = true;
            }
        });
        if (modificato) salvaArriviBaia(arrivi);
    }

    function categoriaBaia(door) {
        if (/^P/i.test(door)) return "Parcheggio";
        const n = parseInt(door, 10);
        if (!isNaN(n)) {
            const inScarico = (n >= 101 && n <= 107) || (n >= 10 && n <= 14) || (n >= 201 && n <= 207) || (n >= 601 && n <= 609) || (n >= 501 && n <= 509);
            if (inScarico) return "Scarico";
            if (n >= 301 && n <= 456) return "Carico";
        }
        return "Freight";
    }

    function elencoSlotCutoffUnici() {
        const visti = new Set();
        const slot = [];
        CATEGORIE.forEach((c) => {
            const chiave = c.cutoffOre + ":" + c.cutoffMin;
            if (!visti.has(chiave)) {
                visti.add(chiave);
                slot.push({ ore: c.cutoffOre, min: c.cutoffMin });
            }
        });
        return slot;
    }
    const SLOT_CUTOFF_UNICI = elencoSlotCutoffUnici();

    function eOrarioDiChiusura(data) {
        const giorno = data.getDay();
        const minutiGiorno = data.getHours() * 60 + data.getMinutes();
        if (giorno === 5 && minutiGiorno >= 9 * 60) return true;
        if (giorno === 6) return true;
        if (giorno === 0 && minutiGiorno < ORARIO_RIAPERTURA_DOMENICA_MINUTI) return true;
        return false;
    }

    function trovaProssimoCutoffConMargine(cutoffBase, arrivoPortaReale, margineMinimo) {
        const ancora = arrivoPortaReale && !isNaN(arrivoPortaReale.getTime()) && arrivoPortaReale.getTime() > cutoffBase.getTime() ? arrivoPortaReale : cutoffBase;
        const candidati = [];
        for (let offset = 0; offset <= 5; offset++) {
            SLOT_CUTOFF_UNICI.forEach((slot) => {
                const d = new Date(ancora.getFullYear(), ancora.getMonth(), ancora.getDate() + offset, slot.ore, slot.min, 0, 0);
                if (d.getTime() > cutoffBase.getTime() && !eOrarioDiChiusura(d)) candidati.push(d);
            });
        }
        candidati.sort((a, b) => a.getTime() - b.getTime());
        for (const d of candidati) {
            const margine = (d.getTime() - arrivoPortaReale.getTime()) / 60000;
            if (margine >= margineMinimo) return d;
        }
        return candidati.length > 0 ? candidati[candidati.length - 1] : null;
    }

    function calcolaCutoffPerCodice(codice, arrivo, arrivoPortaReale) {
        const margineMinimo = margineMinimoPerCodice(codice);
        const candidati = CATEGORIE.filter((c) => c.codici.includes(codice));
        if (candidati.length === 0) return null;
        const choices = [];
        candidati.forEach((cat) => {
            for (let offset = -1; offset <= 5; offset++) {
                const d = new Date(arrivo.getFullYear(), arrivo.getMonth(), arrivo.getDate() + offset, cat.cutoffOre, cat.cutoffMin, 0, 0);
                if (d.getTime() >= arrivo.getTime() && !eOrarioDiChiusura(d)) {
                    choices.push({ cat: cat, cutoffDate: d });
                    break;
                }
            }
        });
        if (choices.length === 0) return null;
        choices.sort((a, b) => a.cutoffDate.getTime() - b.cutoffDate.getTime());
        let risultato = null;
        if (choices.length > 1) {
            for (let i = 0; i < choices.length; i++) {
                const diffMinutiArrivoCutoff = (choices[i].cutoffDate.getTime() - arrivo.getTime()) / 60000;
                if (diffMinutiArrivoCutoff >= margineMinimo) { risultato = choices[i]; break; }
            }
            if (!risultato) risultato = choices[0];
        } else {
            risultato = choices[0];
        }

        if (arrivoPortaReale && !isNaN(arrivoPortaReale.getTime())) {
            const margineReale = (risultato.cutoffDate.getTime() - arrivoPortaReale.getTime()) / 60000;
            if (margineReale < margineMinimo) {
                const nuovoCutoff = trovaProssimoCutoffConMargine(risultato.cutoffDate, arrivoPortaReale, margineMinimo);
                if (nuovoCutoff) {
                    risultato = { cat: risultato.cat, cutoffDate: nuovoCutoff, cutoffOriginale: risultato.cutoffDate, ritardo: true, margineMinimo: margineMinimo };
                }
            }
        }
        if (risultato) risultato.margineMinimo = margineMinimo;
        return risultato;
    }

    function traduciBlocco(el) {
        let textNode = null;
        for (const child of el.childNodes) {
            if (child.nodeType === 3 && child.nodeValue.trim() !== "") {
                textNode = child;
                break;
            }
        }
        if (!textNode) return;
        const code = textNode.nodeValue.trim();
        if (el.getAttribute(CODE_ATTR) === code) return;
        el.setAttribute(CODE_ATTR, code);
        const vecchiaTraduzione = el.querySelector(".tnt-translation");
        if (vecchiaTraduzione) vecchiaTraduzione.remove();
        const parola = codeMap[code];
        if (!parola) return;
        const span = document.createElement("span");
        span.className = "tnt-translation";
        span.textContent = " (" + parola + ")";
        span.style.cssText = "color:#FF7A00; font-weight:bold; font-family:Roboto,sans-serif; font-size:inherit; white-space:nowrap;";
        el.appendChild(span);
    }

    function scanBlocchi(root) {
        if (root.nodeType !== 1) return;
        if (root.matches && root.matches(TARGET_SELECTOR)) traduciBlocco(root);
        if (root.querySelectorAll) root.querySelectorAll(TARGET_SELECTOR).forEach(traduciBlocco);
    }

    function trovaValorePerChiave(root, chiave) {
        const keys = root.querySelectorAll(".door-info__movmt-key");
        for (const k of keys) {

    function estraiCodiceGrezzo(valueEl) {
        if (!valueEl) return null;
        if (valueEl.dataset && valueEl.dataset.tntCode) return valueEl.dataset.tntCode.trim();
        const clone = valueEl.cloneNode(true);
        const trad = clone.querySelector(".tnt-translation");
        if (trad) trad.remove();
        return clone.textContent.trim();
    }

    function parseDataOraSlash(testo) {
        const m = testo.trim().match(/(\d{2})[\/\-](\d{2})\s+(\d{2}):(\d{2})/);
        if (!m) return null;
        const giorno = parseInt(m[1], 10), mese = parseInt(m[2], 10), ore = parseInt(m[3], 10), min = parseInt(m[4], 10);
        return new Date(new Date().getFullYear(), mese - 1, giorno, ore, min);
    }

    function processaTripDetails() {
        const contenitori = document.querySelectorAll(".door-info__movmt-no-left-padding");
        contenitori.forEach((contenitore) => {
            try {
                const tripDetails = contenitore.querySelector(".door-info__trip-details");
                const sezione = contenitore.querySelector(".door-info__movmt-section");
                if (!tripDetails || !sezione) return;
                const selezionato = tripDetails.querySelector(".door-info__header--nav-text--selected");
                if (!selezionato || selezionato.textContent.trim() !== "INBOUND") {
                    const vecchia = sezione.querySelector(".tnt-turno-row");
                    if (vecchia) vecchia.remove();
                    sezione.removeAttribute("data-tnt-turno-key");
                    const vecchiaArrivo = contenitore.querySelector(".tnt-custom-info-row");
                    if (vecchiaArrivo) vecchiaArrivo.remove();
                    return;
                }
                const eBaiaCarico = baiaSelezionata && categoriaBaia(baiaSelezionata) === "Carico";
                if (baiaSelezionata) {
                    const remarksEl = trovaValorePerChiave(sezione, "YOS Remarks") || trovaValorePerChiave(sezione, "Arrival Remarks");
                    const testoRemarks = remarksEl ? (remarksEl.dataset && remarksEl.dataset.tntCode ? remarksEl.dataset.tntCode : remarksEl.textContent.trim()) : "";
                    const eraEscluso = remarksEsclusiOrologio(cacheRemarksPerBaia.get(baiaSelezionata));
                    if (testoRemarks) cacheRemarksPerBaia.set(baiaSelezionata, { testo: testoRemarks, ts: new Date().toISOString() });
                    else cacheRemarksPerBaia.delete(baiaSelezionata);
                    if (eraEscluso !== remarksEsclusiOrologio(cacheRemarksPerBaia.get(baiaSelezionata))) pianificaSincronizzazione(800);
                }
                const origineEl = trovaValorePerChiave(sezione, "Origin");
                const arrivoEl = trovaValorePerChiave(sezione, "Sched. Arrival");
                const actualArrivoEl = trovaValorePerChiave(sezione, "Actual Arrival");
                const tripIdEl = trovaValorePerChiave(sezione, "Sched Trip");
                const tripId = tripIdEl ? tripIdEl.textContent.trim() : null;
                let risultato = null;
                let codiceOrigine = null;
                let testoArrivo = "";
                let testoActualArrivo = "";
                let testoArrivoPorta = "";
                if (origineEl && arrivoEl) {
                    codiceOrigine = estraiCodiceGrezzo(origineEl);
                    testoArrivo = arrivoEl.textContent.trim();
                    if (codiceOrigine && !testoArrivo.toUpperCase().includes("N/A")) {
                        let arrivo = parseDataOraSlash(testoArrivo);
                        if (actualArrivoEl) {
                            testoActualArrivo = actualArrivoEl.textContent.trim();
                            if (testoActualArrivo && !testoActualArrivo.toUpperCase().includes("N/A")) {
                                const arrivoReale = parseDataOraSlash(testoActualArrivo);
                                if (arrivoReale) arrivo = arrivoReale;
                            }
                        }
                        let arrivoPortaReale = null;
                        if (baiaSelezionata) {
                            const arriviSalvati = caricaArriviBaia();
                            const voceBaia = arriviSalvati[baiaSelezionata];
                            if (voceBaia && voceBaia.orario) {
                                const d = new Date(voceBaia.orario);
                                if (!isNaN(d.getTime())) { arrivoPortaReale = d; testoArrivoPorta = voceBaia.orario; }
                            }
                        }
                        if (arrivo && !eBaiaCarico) {
                            risultato = calcolaCutoffPerCodice(codiceOrigine, arrivo, arrivoPortaReale);
                        }
                    }
                }
                if (baiaSelezionata && codiceOrigine && !eBaiaCarico) registraArrivoDaTripDetailsSeMancante(baiaSelezionata, codiceOrigine, tripId, testoActualArrivo, risultato);
                aggiornaRigaArrivoECountdown(contenitore, risultato, eBaiaCarico);
                if (risultato && baiaSelezionata && !eBaiaCarico) cacheCutoffPerBaia.set(baiaSelezionata, risultato);
                if (!origineEl || !arrivoEl) return;
                const chiave = codiceOrigine + "|" + testoArrivo + "|" + testoActualArrivo + "|" + testoArrivoPorta;
                if (sezione.getAttribute("data-tnt-turno-key") === chiave) return;
                sezione.setAttribute("data-tnt-turno-key", chiave);
                const vecchia = sezione.querySelector(".tnt-turno-row");
                if (vecchia) vecchia.remove();
                const vecchioAvviso = sezione.querySelector(".tnt-avviso-ritardo");
                if (vecchioAvviso) vecchioAvviso.remove();
                if (eBaiaCarico) return;
                if (!codiceOrigine || testoArrivo.toUpperCase().includes("N/A")) return;
                const riga = document.createElement("div");
                riga.className = "row door-info__movmt-section-row tnt-turno-row";
                if (risultato) {
                    riga.appendChild(creaColTurno("Turno", risultato.cat.turno));
                    riga.appendChild(creaColTurno("Tipo Cutoff", risultato.cat.nome));
                    const hh = String(risultato.cutoffDate.getHours()).padStart(2, "0");
                    const mm = String(risultato.cutoffDate.getMinutes()).padStart(2, "0");
                    riga.appendChild(creaColTurno("Orario Cutoff", hh + ":" + mm));
                } else {
                    riga.appendChild(creaColTurno("Turno", "N/A"));
                    riga.appendChild(creaColTurno("Tipo Cutoff", "N/A"));
                    riga.appendChild(creaColTurno("Orario Cutoff", "N/A"));
                }
                sezione.appendChild(riga);
                if (!risultato && codiceOrigine && !CATEGORIE.some((c) => c.codici.includes(codiceOrigine))) {
                    const avvisoCodice = document.createElement("div");
                    avvisoCodice.className = "row door-info__movmt-section-row tnt-avviso-ritardo";
                    avvisoCodice.innerHTML = '<div class="col-12"><div class="tnt-banner-ritardo">Codice origine non riconosciuto: <strong>' + escapeHtml(codiceOrigine) + '</strong> \u2014 nessun cutoff calcolabile, va censito</div></div>';
                    sezione.appendChild(avvisoCodice);
                }
                if (risultato && risultato.ritardo && risultato.cutoffOriginale) {
                    const hhO = String(risultato.cutoffOriginale.getHours()).padStart(2, "0");
                    const mmO = String(risultato.cutoffOriginale.getMinutes()).padStart(2, "0");
                    const hhN = String(risultato.cutoffDate.getHours()).padStart(2, "0");
                    const mmN = String(risultato.cutoffDate.getMinutes()).padStart(2, "0");
                    const avviso = document.createElement("div");
                    avviso.className = "row door-info__movmt-section-row tnt-avviso-ritardo";
                    avviso.innerHTML = '<div class="col-12"><div class="tnt-banner-ritardo">Cutoff spostato per arrivo in ritardo: <strong>' + hhO + ':' + mmO + '</strong> &rarr; <strong>' + hhN + ':' + mmN + '</strong> (meno di ' + (risultato.margineMinimo || MARGINE_MINIMO_INTERNAZIONALE_MINUTI) + ' minuti di lavorazione disponibili)</div></div>';
                    sezione.appendChild(avviso);
                }
            } catch (e) {}
        });
    }

    const COLORI_UNIT_PRONTA = ["rgb(135,185,172)"];

    function rigaSegnalataPronta(riga) {
        if (riga.querySelector(".unit_ready_outline") || riga.classList.contains("unit_ready_outline")) return true;
        const dockingBg = riga.querySelector('[id^="dockingBg_"]');
        if (!dockingBg) return false;
        if (dockingBg.className && typeof dockingBg.className === "string" && /ready/i.test(dockingBg.className)) return true;
        let colore = dockingBg.style && dockingBg.style.backgroundColor ? dockingBg.style.backgroundColor : "";
        if (!colore && window.getComputedStyle) {
            try { colore = window.getComputedStyle(dockingBg).backgroundColor || ""; } catch (e) {}
        }
        colore = colore.replace(/\s+/g, "");
        if (!colore) return false;
        return COLORI_UNIT_PRONTA.some((c) => colore.indexOf(c) !== -1);
    }

    function leggiOutboundInScadenza() {
        const righe = document.querySelectorAll(".scheduled-root.outbound-task, .scheduled-root.highlight-task, .scheduled-root.highlight-undeparted-unit");
        const risultati = [];
        const ora = new Date();
        righe.forEach((riga) => {
            const doorEl = riga.querySelector(".outbound-door-number span");
            const dataEl = riga.querySelector(".outbound-dates-position-change");
            const destEl = riga.querySelector(".outbound-current-destination");
            if (!doorEl || !dataEl) return;
            const door = doorEl.textContent.trim();
            const testo = dataEl.textContent.trim();
            if (!testo || testo.toUpperCase().includes("N/A") || door === "-" || !door || categoriaBaia(door) === "Scarico") return;
            if (rigaSegnalataPronta(riga)) return;
            const match = testo.match(/(\d{2})-(\d{2})\s+(\d{2}):(\d{2})/);
            if (!match) return;
            const giorno = parseInt(match[1], 10), mese = parseInt(match[2], 10), ore = parseInt(match[3], 10), minuti = parseInt(match[4], 10);
            const partenza = new Date(ora.getFullYear(), mese - 1, giorno, ore, minuti);
            const pronto = new Date(partenza.getTime() - 20 * 60000);
            const diffMinuti = (pronto.getTime() - ora.getTime()) / 60000;
            if (diffMinuti <= SOGLIA_OUTBOUND_MINUTI && diffMinuti > -60) {
                risultati.push({ door, destinazione: destEl ? destEl.textContent.trim() : "", diffMinuti, categoria: categoriaBaia(door) });
            }
        });
        risultati.sort((a, b) => a.diffMinuti - b.diffMinuti);
        const visti = new Set();
        return risultati.filter((r) => {
            const chiave = r.door + "|" + r.destinazione;
            if (visti.has(chiave)) return false;
            visti.add(chiave);
            return true;
        });
    }

        if (!doorNumber) return;
        const cat = categoriaBaia(doorNumber);
        if (cat !== "Scarico" && cat !== "Parcheggio") return;
        if (!testoActualArrivo || testoActualArrivo.toUpperCase().includes("N/A")) return;
        const dataActual = parseDataOraSlash(testoActualArrivo);
        if (!dataActual) return;
        const orarioFallback = dataActual.toISOString();
        const idFallback = "TRIP_" + (tripId || (codiceOrigine + "_" + orarioFallback));
        const arrivi = caricaArriviBaia();
        const esistente = arrivi[doorNumber];
        if (esistente && String(esistente.unitId).indexOf("TRIP_") !== 0) return;
        if (esistente && esistente.unitId === idFallback) return;
        arrivi[doorNumber] = { unitId: idFallback, orario: orarioFallback, action: "TRIP_DETAILS_FALLBACK" };
        salvaArriviBaia(arrivi);
        cacheMovimentiPerUnit.set(idFallback, {
            origin: codiceOrigine,
            destination: null,
            scheduleArrival: null,
            estimatedArrival: null,
            actualArrival: orarioFallback,
            movementName: tripId || null,
            tripTourDescription: "",
            vistoIl: Date.now()
        });
        if (risultato) cacheCutoffPerBaia.set(doorNumber, risultato);
        registraInStoricoReport(doorNumber, idFallback, orarioFallback);
    }

    function salvaArrayConPulizia(chiave, array, chiaveTempoFn) {
        let arr = array.slice().sort((a, b) => new Date(chiaveTempoFn(b)).getTime() - new Date(chiaveTempoFn(a)).getTime());
        let scartati = false;
        while (arr.length > 0) {
            try {
                localStorage.setItem(chiave, JSON.stringify(arr));
                if (scartati) {
                    console.warn("[YOS TOOL] Quota locale piena: rimossi i " + (array.length - arr.length) + " record piu' vecchi di", chiave);
                    avvisaStorageQuasiPieno();
                }
                return arr;
            } catch (e) {
                scartati = true;
                const daRimuovere = Math.max(1, Math.floor(arr.length * 0.1));
                arr = arr.slice(0, arr.length - daRimuovere);
            }
        }
        try { localStorage.removeItem(chiave); } catch (e) {}
        avvisaStorageLocalePieno();
        return arr;
    }

    function gmRequest(dettagli) {
        return new Promise((resolve, reject) => {
            if (typeof GM_xmlhttpRequest !== "function") { reject(new Error("Rete: GM_xmlhttpRequest non disponibile")); return; }
            GM_xmlhttpRequest(Object.assign({}, dettagli, {
                timeout: 15000,
                onload: (res) => resolve(res),
                onerror: () => reject(new Error("Rete: richiesta fallita")),
                ontimeout: () => reject(new Error("Rete: timeout")),
                onabort: () => reject(new Error("Rete: richiesta interrotta"))
            }));
        });
    }

    function estraiETag(res) {
        const raw = res && res.responseHeaders ? res.responseHeaders : "";
        const m = /etag:\s*"?([^"\r\n]+?)"?\s*$/im.exec(raw);
        return m ? m[1] : null;
    }

    async function firebaseCreaSessioneAnonima() {
        const res = await gmRequest({
            method: "POST",
            url: "https://identitytoolkit.googleapis.com/v1/accounts:signUp?key=" + FIREBASE_SYNC.WEB_API_KEY,
            headers: { "Content-Type": "application/json" },
            data: JSON.stringify({ returnSecureToken: true })
        });
        if (res.status < 200 || res.status >= 300) throw new Error("Firebase Auth signUp " + res.status);
        const dati = JSON.parse(res.responseText);
        const auth = { idToken: dati.idToken, refreshToken: dati.refreshToken, scadenza: Date.now() + Number(dati.expiresIn) * 1000 };
        scriviAuthSalvato(auth);
        return auth;
    }

    async function firebaseRinnovaSessione(refreshToken) {
        const res = await gmRequest({
            method: "POST",
            url: "https://securetoken.googleapis.com/v1/token?key=" + FIREBASE_SYNC.WEB_API_KEY,
            headers: { "Content-Type": "application/x-www-form-urlencoded" },
            data: "grant_type=refresh_token&refresh_token=" + encodeURIComponent(refreshToken)
        });
        if (res.status < 200 || res.status >= 300) throw new Error("Firebase Auth refresh " + res.status);
        const dati = JSON.parse(res.responseText);
        const auth = { idToken: dati.id_token, refreshToken: dati.refresh_token, scadenza: Date.now() + Number(dati.expires_in) * 1000 };
        scriviAuthSalvato(auth);
        return auth;
    }

    async function otteniIdTokenValido() {
        let auth = leggiAuthSalvato();
        if (auth && auth.scadenza > Date.now() + 60000) return auth.idToken;
        try {
            if (auth && auth.refreshToken) auth = await firebaseRinnovaSessione(auth.refreshToken);
            else auth = await firebaseCreaSessioneAnonima();
        } catch (e) {
            auth = await firebaseCreaSessioneAnonima();
        }
        return auth.idToken;
    }

    async function firebaseLeggiFile() {
        const idToken = await otteniIdTokenValido();
        const url = FIREBASE_SYNC.BASE_URL + "/" + FIREBASE_SYNC.PATH + "?auth=" + idToken;
        const res = await gmRequest({
            method: "GET",
            url,
            headers: { "X-Firebase-ETag": "true" }
        });
        if (res.status < 200 || res.status >= 300) throw new Error("Firebase GET " + res.status);
        let dati = null;
        try { dati = JSON.parse(res.responseText); } catch (e) { dati = null; }
        if (!dati || typeof dati !== "object") dati = { versione: VERSIONE_DATI, storico: [], sessioni: [], presenze: {}, remarksBaie: {} };
        if (!Array.isArray(dati.storico)) dati.storico = [];
        if (!Array.isArray(dati.sessioni)) dati.sessioni = [];
        if (!dati.presenze || typeof dati.presenze !== "object") dati.presenze = {};
        if (!dati.remarksBaie || typeof dati.remarksBaie !== "object") dati.remarksBaie = {};
        if (typeof dati.versione !== "number") dati.versione = 0;
        return { dati, etag: estraiETag(res) };
    }

    async function firebaseScriviFile(dati, etag) {
        const idToken = await otteniIdTokenValido();
        const url = FIREBASE_SYNC.BASE_URL + "/" + FIREBASE_SYNC.PATH + "?auth=" + idToken;
        const headers = { "Content-Type": "application/json" };
        if (etag) headers["if-match"] = etag;
        const res = await gmRequest({
            method: "PUT",
            url,
            headers,
            data: JSON.stringify(dati)
        });
        if (res.status === 412) return false;
        if (res.status < 200 || res.status >= 300) throw new Error("Firebase PUT " + res.status);
        return true;
    }

    let sincronizzazioneInCorso = false;
    let sincronizzazionePendente = false;
    let timerSincronizzazione = null;

    function pianificaSincronizzazione(ritardoMs) {
        if (!FIREBASE_SYNC.ATTIVO) return;
        if (timerSincronizzazione) clearTimeout(timerSincronizzazione);
        timerSincronizzazione = setTimeout(sincronizzaDati, ritardoMs || 5000);
    }

    function attesa(ms) { return new Promise((r) => setTimeout(r, ms)); }

    function unisciRemarks(remotoObj) {
        const limite = Date.now() - SCADENZA_REMARKS_ORE * 60 * 60 * 1000;
        const combinate = new Map();
        Object.keys(remotoObj || {}).forEach((k) => {
            const v = remotoObj[k];
            if (v && v.testo && v.ts) combinate.set(k, v);
        });
        cacheRemarksPerBaia.forEach((v, k) => {
            const esistente = combinate.get(k);
            if (!esistente || new Date(v.ts).getTime() >= new Date(esistente.ts).getTime()) combinate.set(k, v);
        });
        combinate.forEach((v, k) => { if (new Date(v.ts).getTime() < limite) combinate.delete(k); });
        return combinate;
    }

    async function sincronizzaDati() {
        if (!FIREBASE_SYNC.ATTIVO) { aggiornaStatoSync(false, null); return; }
        if (sincronizzazioneInCorso) { sincronizzazionePendente = true; return; }
        sincronizzazioneInCorso = true;
        try {
            let presenzeFinali = {};
            for (let tentativo = 0; tentativo < 6; tentativo++) {
                const remoto = await firebaseLeggiFile();
                if (remoto.dati.versione > VERSIONE_DATI) {
                    console.warn("[YOS TOOL] Formato dati remoto piu' recente di questo script: sync sospesa");
                    aggiornaStatoSync(false, null);
                    return;
                }
                const storicoUnito = unisciSenzaDuplicati(remoto.dati.storico, caricaStoricoReport(), chiaveStorico);
                const sessioniUnite = unisciSenzaDuplicati(remoto.dati.sessioni, caricaSessioniScarico(), chiaveSessione);
                const pulito = applicaRetentionCondivisa(storicoUnito, sessioniUnite);
                presenzeFinali = aggiornaPresenze(remoto.dati.presenze);

                const remarksUnite = unisciRemarks(remoto.dati.remarksBaie);
                cacheRemarksPerBaia.clear();
                remarksUnite.forEach((v, k) => cacheRemarksPerBaia.set(k, v));
                const remarksOggetto = {};
                remarksUnite.forEach((v, k) => { remarksOggetto[k] = v; });

                salvaStoricoReport(pulito.storico);
                salvaSessioniScarico(pulito.sessioni);

                const invariato = pulito.storico.length === remoto.dati.storico.length
                    && pulito.sessioni.length === remoto.dati.sessioni.length
                    && remoto.dati.presenze[DEVICE_ID] === presenzeFinali[DEVICE_ID]
                    && Object.keys(presenzeFinali).length === Object.keys(remoto.dati.presenze).length
                    && JSON.stringify(remarksOggetto) === JSON.stringify(remoto.dati.remarksBaie || {});
                if (invariato) break;

                const ok = await firebaseScriviFile({ versione: VERSIONE_DATI, storico: pulito.storico, sessioni: pulito.sessioni, presenze: presenzeFinali, remarksBaie: remarksOggetto }, remoto.etag);
                if (ok) break;
                await attesa(400 * (tentativo + 1) + Math.random() * 300);
            }
            aggiornaStatoSync(true, contaUtentiOggi(presenzeFinali));
        } catch (e) {
            console.warn("[YOS TOOL] Sincronizzazione fallita:", e.message);
            const motivo = /\s(401|403)\b/.test(e.message || "") ? "token non valido" : null;
            aggiornaStatoSync(false, null, motivo);
        } finally {
            sincronizzazioneInCorso = false;
            if (sincronizzazionePendente) { sincronizzazionePendente = false; pianificaSincronizzazione(2000); }
        }
    }

    function pulisciStoricoVecchio() {
        const storico = caricaStoricoReport();
        const limite = Date.now() - GIORNI_STORICO_TREND * 24 * 60 * 60 * 1000;
        if (storico.length > 0) {
            const filtrato = storico.filter((r) => {
                const t = r.timestampIso ? new Date(r.timestampIso).getTime() : NaN;
                return !isNaN(t) && t >= limite;
            });
            if (filtrato.length !== storico.length) salvaStoricoReport(filtrato);
        }
        const sessioni = caricaSessioniScarico();
        if (sessioni.length > 0) {
            const sessioniFiltrate = sessioni.filter((s) => {
                const t = s.inizioIso ? new Date(s.inizioIso).getTime() : NaN;
                return !isNaN(t) && t >= limite;
            });
            if (sessioniFiltrate.length !== sessioni.length) salvaSessioniScarico(sessioniFiltrate);
        }
    }

    function registraInStoricoReport(doorNumber, unitId, orarioIso) {
        pulisciStoricoVecchio();
        const storico = caricaStoricoReport();
        const esisteGia = storico.some(item => item.doorNumber === doorNumber && item.unitId === unitId);
        if (esisteGia) return;

        const dataArrivo = new Date(orarioIso);
        const orarioFormattato = String(dataArrivo.getHours()).padStart(2, "0") + ":" + String(dataArrivo.getMinutes()).padStart(2, "0");
        const dataFormattata = String(dataArrivo.getDate()).padStart(2, "0") + "/" + String(dataArrivo.getMonth() + 1).padStart(2, "0");

        let codiceOrigine = "N/A";
        let localita = "N/A";
        let viaggio = "N/A";
        let destinazione = "N/A";
        let arrivoSchedulato = "N/A";
        let arrivoStimato = "N/A";
        let arrivoReale = "N/A";
        let cutoffOrario = "N/A";
        let cutoffTurno = "N/A";
        let cutoffTipo = "N/A";
        let ritardoFlag = "No";
        let cutoffOriginaleTesto = "";

        function formattaOra(iso) {
            if (!iso) return "N/A";
            const d = new Date(iso);
            if (isNaN(d.getTime())) return "N/A";
            return String(d.getDate()).padStart(2, "0") + "/" + String(d.getMonth() + 1).padStart(2, "0") + " " + String(d.getHours()).padStart(2, "0") + ":" + String(d.getMinutes()).padStart(2, "0");
        }

        if (unitId) {
            const mov = cacheMovimentiPerUnit.get(String(unitId));
            if (mov) {
                codiceOrigine = mov.origin || "N/A";
                localita = codeMap[codiceOrigine] || "N/A";
                viaggio = mov.movementName || "N/A";
                if (mov.tripTourDescription) viaggio += " (" + mov.tripTourDescription + ")";
                destinazione = mov.destination || "N/A";
                arrivoSchedulato = formattaOra(mov.scheduleArrival);
                arrivoStimato = formattaOra(mov.estimatedArrival);
                arrivoReale = formattaOra(mov.actualArrival);

                const arrivoIso = mov.actualArrival || mov.scheduleArrival || mov.estimatedArrival;
                if (arrivoIso) {
                    const arrivoDate = new Date(arrivoIso);
                    const arrivoPortaReale = new Date(orarioIso);
                    const res = calcolaCutoffPerCodice(codiceOrigine, arrivoDate, !isNaN(arrivoPortaReale.getTime()) ? arrivoPortaReale : null);
                    if (res) {
                        cutoffOrario = String(res.cutoffDate.getHours()).padStart(2, "0") + ":" + String(res.cutoffDate.getMinutes()).padStart(2, "0");
                        cutoffTurno = res.cat.turno;
                        cutoffTipo = res.cat.nome;
                        if (res.ritardo && res.cutoffOriginale) {
                            ritardoFlag = "Si";
                            cutoffOriginaleTesto = String(res.cutoffOriginale.getHours()).padStart(2, "0") + ":" + String(res.cutoffOriginale.getMinutes()).padStart(2, "0");
                        }
                    }
                }
            }
        }

        storico.push({
            data: dataFormattata,
            timestampIso: orarioIso,
            doorNumber,
            unitId: unitId || "N/A",
            orarioArrivo: orarioFormattato,
            codiceOrigine,
            localita,
            viaggio,
            destinazione,
            arrivoSchedulato,
            arrivoStimato,
            arrivoReale,
            cutoffOrario,
            cutoffTurno,
            cutoffTipo,
            ritardoFlag,
            cutoffOriginaleTesto
        });
        salvaStoricoReport(storico);
        pianificaSincronizzazione(5000);
    }

    function costruisciDataOra(dataStr, oraStr, oraFallback) {
        if (!dataStr) return null;
        const parti = dataStr.split("-").map(Number);
        const anno = parti[0], mese = parti[1], giorno = parti[2];
        const oraEffettiva = oraStr || oraFallback || "00:00";
        const partiOra = oraEffettiva.split(":").map(Number);
        const ore = partiOra[0] || 0, minuti = partiOra[1] || 0;
        return new Date(anno, mese - 1, giorno, ore, minuti, 0, 0);
    }

    function filtraPerPeriodoConOra(lista, dataInizioStr, oraInizioStr, dataFineStr, oraFineStr, campoData) {
        campoData = campoData || "timestampIso";
        if (!dataInizioStr && !dataFineStr) return lista;
        const inizio = dataInizioStr ? costruisciDataOra(dataInizioStr, oraInizioStr, "00:00") : null;
        const fine = dataFineStr ? costruisciDataOra(dataFineStr, oraFineStr, "23:59") : null;
        return lista.filter((r) => {
            const iso = r[campoData];
            if (!iso) return true;
            const d = new Date(iso);
            if (isNaN(d.getTime())) return true;
            if (inizio && d < inizio) return false;
            if (fine && d > fine) return false;
            return true;
        });
    }

    function filtraPerPeriodo(lista, dataInizioStr, dataFineStr, campoData) {
        campoData = campoData || "timestampIso";
        if (!dataInizioStr && !dataFineStr) return lista;
        const inizio = dataInizioStr ? new Date(dataInizioStr + "T00:00:00") : null;
        const fine = dataFineStr ? new Date(dataFineStr + "T23:59:59") : null;
        return lista.filter((r) => {
            const iso = r[campoData];
            if (!iso) return true;
            const d = new Date(iso);
            if (isNaN(d.getTime())) return true;
            if (inizio && d < inizio) return false;
            if (fine && d > fine) return false;
            return true;
        });
    }

    async function scaricaReportExcel(dataInizioStr, dataFineStr) {
        const storicoCompleto = caricaStoricoReport();
        if (storicoCompleto.length === 0) {
            alert("Nessun dato registrato per il report.");
            return;
        }
        const storico = filtraPerPeriodo(storicoCompleto, dataInizioStr, dataFineStr, "timestampIso");
        if (storico.length === 0) {
            alert("Nessun dato registrato nel periodo selezionato.");
            return;
        }

        let XLSX;
        try { XLSX = await caricaLibreriaXLSX(); } catch (e) { alert(e.message); return; }

        const righeStorico = storico.map((r) => ({
            "Data": r.data,
            "Timestamp": r.timestampIso,
            "Baia": r.doorNumber,
            "Categoria Baia": categoriaBaia(r.doorNumber),
            "Lato": latoBaia(r.doorNumber) || "",
            "Unit ID": r.unitId,
            "Ora Arrivo": r.orarioArrivo,
            "Codice Origine": r.codiceOrigine,
            "Localita (tradotta)": r.localita,
            "Info Viaggio": r.viaggio,
            "Destinazione": r.destinazione || "N/A",
            "Arrivo Schedulato": r.arrivoSchedulato || "N/A",
            "Arrivo Stimato": r.arrivoStimato || "N/A",
            "Arrivo Reale": r.arrivoReale || "N/A",
            "Ora Cutoff": r.cutoffOrario,
            "Turno": r.cutoffTurno,
            "Categoria Cutoff": r.cutoffTipo,
            "In Ritardo": r.ritardoFlag || "No",
            "Cutoff Originale (se slittato)": r.cutoffOriginaleTesto || ""
        }));

        const sessioni = filtraPerPeriodo(caricaSessioniScarico(), dataInizioStr, dataFineStr, "inizioIso");
        const righeSessioni = sessioni.map((s) => ({
            "Lato": s.lato,
            "Baia": s.doorNumber,
            "Unit ID": s.unitId,
            "Inizio Scarico": s.inizioIso,
            "Fine Scarico": s.fineIso,
            "Durata (minuti)": s.durataMinuti
        }));

        const stat = calcolaStatisticheTurno(dataInizioStr, null, dataFineStr, null);
        const righeStat = [];
        righeStat.push({ Indicatore: "Camion presi in carico nel periodo (Scarico)", Valore: stat.camionPresiInCarico, Campioni: "" });
        righeStat.push({ Indicatore: "Attesa media in parcheggio (minuti)", Valore: stat.attesaParcheggio.mediaMinuti !== null ? stat.attesaParcheggio.mediaMinuti : "", Campioni: stat.attesaParcheggio.campioni });
        stat.perLato.forEach((r) => righeStat.push({ Indicatore: "Tempo medio scarico - " + r.chiave + " (minuti)", Valore: r.mediaMinuti !== null ? r.mediaMinuti : "", Campioni: r.campioni }));
        stat.perTurno.forEach((r) => righeStat.push({ Indicatore: "% ritardo turno - " + r.chiave, Valore: r.percentuale + "%", Campioni: r.totale }));
        stat.perCliente.forEach((r) => righeStat.push({ Indicatore: "% ritardo cliente - " + r.chiave, Valore: r.percentuale + "%", Campioni: r.totale }));

        const righeStatBaia = stat.perBaia.map((r) => ({
            "Baia": r.doorNumber,
            "Lato": r.lato,
            "Media Minuti": r.mediaMinuti,
            "Campioni": r.campioni,
            "Scarto vs Media Lato (min)": r.scartoVsLato !== null ? r.scartoVsLato : ""
        }));

        const wb = XLSX.utils.book_new();
        XLSX.utils.book_append_sheet(wb, XLSX.utils.json_to_sheet(righeStorico), "Storico Arrivi");
        XLSX.utils.book_append_sheet(wb, XLSX.utils.json_to_sheet(righeSessioni), "Sessioni Scarico per Lato");
        XLSX.utils.book_append_sheet(wb, XLSX.utils.json_to_sheet(righeStat), "Riepilogo Statistiche");
        XLSX.utils.book_append_sheet(wb, XLSX.utils.json_to_sheet(righeStatBaia), "Statistiche per Baia");

        const suffisso = (dataInizioStr || dataFineStr)
            ? `_${dataInizioStr || "inizio"}_a_${dataFineStr || "fine"}`
            : `_TUTTO_${new Date().toISOString().slice(0,10)}`;
        XLSX.writeFile(wb, `REPORT_TURNO_YOS${suffisso}.xlsx`);
    }


    function filtraStoricoPerPeriodo(storico, dataInizioStr, dataFineStr) {
        return filtraPerPeriodo(storico, dataInizioStr, dataFineStr, "timestampIso");

    function calcolaAttesaMediaParcheggio(storico) {
        const perUnit = new Map();
        storico.forEach((r) => {
            if (!r.unitId || r.unitId === "N/A") return;
            const cat = categoriaBaia(r.doorNumber);
            if (cat !== "Parcheggio" && cat !== "Scarico") return;
            const t = r.timestampIso ? new Date(r.timestampIso).getTime() : NaN;
            if (isNaN(t)) return;
            if (!perUnit.has(r.unitId)) perUnit.set(r.unitId, []);
            perUnit.get(r.unitId).push({ cat, t });
        });
        const attese = [];
        perUnit.forEach((eventi) => {
            eventi.sort((a, b) => a.t - b.t);
            const primoParcheggio = eventi.find((e) => e.cat === "Parcheggio");
            if (!primoParcheggio) return;
            const primaBaiaDopo = eventi.find((e) => e.cat === "Scarico" && e.t > primoParcheggio.t);
            if (!primaBaiaDopo) return;
            const minuti = (primaBaiaDopo.t - primoParcheggio.t) / 60000;
            if (minuti >= 0 && minuti < 24 * 60) attese.push(minuti);
        });
        if (attese.length === 0) return { mediaMinuti: null, campioni: 0 };
        const media = attese.reduce((s, v) => s + v, 0) / attese.length;
        return { mediaMinuti: Math.round(media), campioni: attese.length };
    }

    function calcolaTempoScaricoPerLato(dataInizioStr, oraInizioStr, dataFineStr, oraFineStr) {
        const sessioni = filtraPerPeriodoConOra(caricaSessioniScarico(), dataInizioStr, oraInizioStr, dataFineStr, oraFineStr, "inizioIso");
        const ordineLati = ["Lato 100", "Lato Furgoni", "Lato 200", "Lato 600", "Lato 500"];
        const gruppi = new Map();
        ordineLati.forEach((l) => gruppi.set(l, []));
        sessioni.forEach((s) => {
            if (!s.lato || typeof s.durataMinuti !== "number" || isNaN(s.durataMinuti)) return;
            if (!gruppi.has(s.lato)) gruppi.set(s.lato, []);
            gruppi.get(s.lato).push(s.durataMinuti);
        });
        const righe = [];
        ordineLati.forEach((lato) => {
            const valori = gruppi.get(lato) || [];
            if (valori.length === 0) { righe.push({ chiave: lato, mediaMinuti: null, campioni: 0 }); return; }
            const media = valori.reduce((s, v) => s + v, 0) / valori.length;
            righe.push({ chiave: lato, mediaMinuti: Math.round(media), campioni: valori.length });
        });
        return righe;
    }

    function calcolaTempoScaricoPerBaia(dataInizioStr, oraInizioStr, dataFineStr, oraFineStr) {
        const sessioni = filtraPerPeriodoConOra(caricaSessioniScarico(), dataInizioStr, oraInizioStr, dataFineStr, oraFineStr, "inizioIso");
        const gruppi = new Map();
        sessioni.forEach((s) => {
            if (!s.doorNumber || typeof s.durataMinuti !== "number" || isNaN(s.durataMinuti)) return;
            if (!gruppi.has(s.doorNumber)) gruppi.set(s.doorNumber, []);
            gruppi.get(s.doorNumber).push(s.durataMinuti);
        });
        const medieLato = new Map();
        calcolaTempoScaricoPerLato(dataInizioStr, oraInizioStr, dataFineStr, oraFineStr).forEach((r) => medieLato.set(r.chiave, r.mediaMinuti));
        const righe = [];
        gruppi.forEach((valori, doorNumber) => {
            const media = valori.reduce((s, v) => s + v, 0) / valori.length;
            const lato = latoBaia(doorNumber);
            const mediaLato = lato ? medieLato.get(lato) : null;
            const scartoVsLato = (mediaLato !== null && mediaLato !== undefined) ? Math.round(media - mediaLato) : null;
            righe.push({ doorNumber, lato: lato || "", mediaMinuti: Math.round(media), campioni: valori.length, scartoVsLato });
        });
        righe.sort((a, b) => b.mediaMinuti - a.mediaMinuti);
        return righe;
    }

    function calcolaFinestraTrend(dataInizioStr) {
        if (!dataInizioStr) return null;
        const inizioPeriodo = new Date(dataInizioStr + "T00:00:00");
        if (isNaN(inizioPeriodo.getTime())) return null;
        const fineTrend = new Date(inizioPeriodo.getTime() - 24 * 60 * 60 * 1000);
        const inizioTrend = new Date(fineTrend.getTime() - (GIORNI_MEDIA_TREND - 1) * 24 * 60 * 60 * 1000);
        return { dataInizioStr: inizioTrend.toISOString().slice(0, 10), dataFineStr: fineTrend.toISOString().slice(0, 10) };
    }

    function calcolaTrend(dataInizioStr) {
        const finestra = calcolaFinestraTrend(dataInizioStr);
        if (!finestra) return null;
        const perLato = calcolaTempoScaricoPerLato(finestra.dataInizioStr, "00:00", finestra.dataFineStr, "23:59");
        const mappaLato = new Map();
        perLato.forEach((r) => mappaLato.set(r.chiave, r.mediaMinuti));
        const storicoFinestra = filtraPerPeriodoConOra(caricaStoricoReport(), finestra.dataInizioStr, "00:00", finestra.dataFineStr, "23:59", "timestampIso");
        const attesaParcheggio = calcolaAttesaMediaParcheggio(storicoFinestra);
        return { perLato: mappaLato, attesaParcheggioMediaMinuti: attesaParcheggio.mediaMinuti, giorni: GIORNI_MEDIA_TREND };
    }

    function calcolaRitardiPer(storico, chiaveFn) {
        const gruppi = new Map();
        storico.forEach((r) => {
            const chiave = chiaveFn(r);
            if (!chiave || chiave === "N/A" || chiave === "N/D") return;
            if (!gruppi.has(chiave)) gruppi.set(chiave, { totale: 0, ritardati: 0 });
            const g = gruppi.get(chiave);
            g.totale++;
            if (r.ritardoFlag === "Si") g.ritardati++;
        });
        const righe = [];
        gruppi.forEach((v, chiave) => {
            righe.push({ chiave, totale: v.totale, ritardati: v.ritardati, percentuale: v.totale > 0 ? Math.round((v.ritardati / v.totale) * 100) : 0 });
        });
        righe.sort((a, b) => b.totale - a.totale);
        return righe;
    }

    function calcolaStatisticheTurno(dataInizioStr, oraInizioStr, dataFineStr, oraFineStr) {
        let storico = caricaStoricoReport();
        storico = filtraPerPeriodoConOra(storico, dataInizioStr, oraInizioStr, dataFineStr, oraFineStr, "timestampIso");
        const storicoScaricoParcheggio = storico.filter((r) => {
            const cat = categoriaBaia(r.doorNumber);
            return cat === "Scarico" || cat === "Parcheggio";
        });
        const camionPresiInCarico = storico.filter((r) => categoriaBaia(r.doorNumber) === "Scarico").length;
        return {
            totaleRecord: storicoScaricoParcheggio.length,
            camionPresiInCarico,
            attesaParcheggio: calcolaAttesaMediaParcheggio(storico),
            perLato: calcolaTempoScaricoPerLato(dataInizioStr, oraInizioStr, dataFineStr, oraFineStr),
            perBaia: calcolaTempoScaricoPerBaia(dataInizioStr, oraInizioStr, dataFineStr, oraFineStr),
            perTurno: calcolaRitardiPer(storicoScaricoParcheggio, (r) => r.cutoffTurno),
            perCliente: calcolaRitardiPer(storicoScaricoParcheggio, (r) => r.codiceOrigine),
            trend: calcolaTrend(dataInizioStr)
        };
    }

    function rigaDeltaTrend(deltaMinuti, giorni, mediaStoricaMinuti) {
        if (deltaMinuti === null || deltaMinuti === undefined || isNaN(deltaMinuti)) return "";
        const classe = deltaMinuti > 0 ? "tnt-delta-rosso" : (deltaMinuti < 0 ? "tnt-delta-verde" : "tnt-delta-neutro");
        const segno = deltaMinuti > 0 ? "+" : "";
        return '<div class="tnt-card-delta ' + classe + '">' + segno + deltaMinuti + 'm rispetto alla media ultimi ' + giorni + ' giorni (' + mediaStoricaMinuti + 'm)</div>';
    }

    function trovaRigaPerChiavi(ambito, chiavi) {
        const keys = ambito.querySelectorAll(".door-info__movmt-key");
        for (const k of keys) {
            if (chiavi.includes(k.textContent.trim())) return k.closest(".row") || k.closest(".door-info__movmt-section-row");
        }
        return null;
    }

    function aggiornaRigaArrivoECountdown(ambito, risultato, eBaiaCarico) {
        const vecchia = ambito.querySelector(".tnt-custom-info-row");
        if (vecchia) vecchia.remove();
        const rigaRemarks = trovaRigaPerChiavi(ambito, ["Arrival Remarks", "YOS Remarks"]);
        if (!rigaRemarks) return;
        let valoreArrivo = "N/A";
        if (baiaSelezionata) {
            const arrivi = caricaArriviBaia();
            const voce = arrivi[baiaSelezionata];
            if (voce && voce.orario) {
                const dataArrivo = new Date(voce.orario);
                valoreArrivo = String(dataArrivo.getHours()).padStart(2, "0") + ":" + String(dataArrivo.getMinutes()).padStart(2, "0");
            }
        }
        const rigaNuova = document.createElement("div");
        rigaNuova.className = "row door-info__movmt-section-row tnt-custom-info-row";
        rigaNuova.style.marginTop = "10px";

        if (eBaiaCarico) {
            rigaNuova.appendChild(creaColTurno("Orario Arrivo Camion", valoreArrivo, "col-12"));
            rigaRemarks.insertAdjacentElement("afterend", rigaNuova);
            return;
        }

        let valoreCountdown = "N/A";
        if (risultato && risultato.cutoffDate) {
            const diffMs = risultato.cutoffDate.getTime() - new Date().getTime();
            if (diffMs > 0) {
                const diffMinsTot = Math.floor(diffMs / 60000);
                const ore = Math.floor(diffMinsTot / 60);
                const minuti = diffMinsTot % 60;
                valoreCountdown = ore > 0 ? `${ore}h ${minuti}m` : `${minuti}m`;
            } else {
                const diffMinsTot = Math.floor(Math.abs(diffMs) / 60000);
                const ore = Math.floor(diffMinsTot / 60);
                const minuti = diffMinsTot % 60;
                valoreCountdown = `In ritardo di ${ore > 0 ? ore + 'h ' : ''}${minuti}m`;
            }
        }
        const colArrivo = creaColTurno("Orario Arrivo Camion", valoreArrivo, "col-6");
        const colCountdown = creaColTurno("Countdown Cutoff", valoreCountdown, "col-6");
        const valSpan = colCountdown.querySelector(".door-info__movmt-value");
        if (valSpan && risultato && risultato.cutoffDate) {
            const diffMs = risultato.cutoffDate.getTime() - new Date().getTime();
            if (diffMs < 0) valSpan.style.color = "#ff3b30";
            else if (diffMs <= 30 * 60000) valSpan.style.color = "#ffcc00";
            else valSpan.style.color = "#4cd964";
        }
        rigaNuova.appendChild(colArrivo);
        rigaNuova.appendChild(colCountdown);
        rigaRemarks.insertAdjacentElement("afterend", rigaNuova);
    }

    function leggiCutoffBaiaCorrente() {
        const originClassEl = document.querySelector(".inbound-origin-en-route, .inbound-origin");
        let codiceOrigine = null;
        if (originClassEl && originClassEl.textContent.trim()) codiceOrigine = estraiCodiceGrezzo(originClassEl);
        const contenitori = document.querySelectorAll(".door-info__movmt-no-left-padding");
        for (const contenitore of contenitori) {
            const sezione = contenitore.querySelector(".door-info__movmt-section");
            if (!sezione) continue;
            if (!codiceOrigine) codiceOrigine = estraiCodiceGrezzo(trovaValorePerChiave(sezione, "Origin"));
            const arrivoEl = trovaValorePerChiave(sezione, "Sched. Arrival");
            if (!arrivoEl || !codiceOrigine) continue;
            const testoArrivo = arrivoEl.textContent.trim();
            if (testoArrivo.toUpperCase().includes("N/A")) continue;
            let arrivo = parseDataOraSlash(testoArrivo);
            if (!arrivo) continue;
            const actualArrivoEl = trovaValorePerChiave(sezione, "Actual Arrival");
            if (actualArrivoEl) {
                const testoActual = actualArrivoEl.textContent.trim();
                if (testoActual && !testoActual.toUpperCase().includes("N/A")) {
                    const arrivoReale = parseDataOraSlash(testoActual);
                    if (arrivoReale) arrivo = arrivoReale;
                }
            }
            let arrivoPortaReale = null;
            if (baiaSelezionata) {
                const arriviSalvati = caricaArriviBaia();
                const voceBaia = arriviSalvati[baiaSelezionata];
                if (voceBaia && voceBaia.orario) {
                    const d = new Date(voceBaia.orario);
                    if (!isNaN(d.getTime())) arrivoPortaReale = d;
                }
            }
            const cutoffRes = calcolaCutoffPerCodice(codiceOrigine, arrivo, arrivoPortaReale);
            if (cutoffRes) return { cutoff: cutoffRes, codice: codiceOrigine };
        }
        return null;
    }

    function scansionaBaieScarico(panelBody) {
        const gruppi = new Map();
        const ora = new Date();
        function aggiungiAlGruppo(risultato, infoBaia) {
            const diffMinuti = (risultato.cutoffDate.getTime() - ora.getTime()) / 60000;
            if (diffMinuti > SOGLIA_SCARICO_MINUTI) return;
            const senzaBaiaAssegnata = /^\(.*\)$/.test(infoBaia);
            if (senzaBaiaAssegnata) return;
            const hh = String(risultato.cutoffDate.getHours()).padStart(2, "0");
            const mm = String(risultato.cutoffDate.getMinutes()).padStart(2, "0");
            const dataChiave = risultato.cutoffDate.getFullYear() + "-" + String(risultato.cutoffDate.getMonth() + 1).padStart(2, "0") + "-" + String(risultato.cutoffDate.getDate()).padStart(2, "0");
            const chiave = risultato.cat.nome + "|" + dataChiave + " " + hh + ":" + mm;
            if (!gruppi.has(chiave)) gruppi.set(chiave, { nome: risultato.cat.nome, turno: risultato.cat.turno, orario: risultato.cutoffDate, baie: [] });
            gruppi.get(chiave).baie.push(risultato.ritardo ? infoBaia + "*" : infoBaia);
        }

        const baie = trovaBaieScaricoOccupate();
        const arriviCorrenti = caricaArriviBaia();
        for (const baia of baie) {
            if (remarksEsclusiOrologio(cacheRemarksPerBaia.get(baia.doorNumber))) continue;
            const voceBaia = arriviCorrenti[baia.doorNumber];
            if (categoriaBaia(baia.doorNumber) === "Parcheggio" && voceBaia && unitGiaScaricati.has(String(voceBaia.unitId))) continue;
            let infoCutoff = calcolaCutoffDaCacheMovimenti(baia.doorNumber);
            if (!infoCutoff) {
                const cutoffCachato = cacheCutoffPerBaia.get(baia.doorNumber);
                if (cutoffCachato) infoCutoff = { cutoff: cutoffCachato, codice: null };
            }
            if (!infoCutoff && baiaSelezionata === baia.doorNumber) {
                assicuraTabInbound();
                infoCutoff = leggiCutoffBaiaCorrente();
                if (infoCutoff) cacheCutoffPerBaia.set(baia.doorNumber, infoCutoff.cutoff);
            }
            if (!infoCutoff) continue;
            aggiungiAlGruppo(infoCutoff.cutoff, baia.doorNumber);
        }

        renderPannelloScaricoConteggi(panelBody, gruppi);
    }

    function elencoBaieScaricoComplete() {
        const elenco = [];
        for (let n = 101; n <= 107; n++) elenco.push(String(n));
        for (let n = 10; n <= 14; n++) elenco.push(String(n));
        for (let n = 201; n <= 207; n++) elenco.push(String(n));
        for (let n = 601; n <= 609; n++) elenco.push(String(n));
        for (let n = 501; n <= 509; n++) elenco.push(String(n));
        return elenco;
    }

    function elencoParcheggioScaricoOccupati() {
        const arrivi = caricaArriviBaia();
        const risultati = [];
        Object.keys(arrivi).forEach((doorNumber) => {
            const voce = arrivi[doorNumber];
            if (!voce || !voce.unitId) return;
            if (categoriaBaia(doorNumber) !== "Parcheggio") return;
            if (unitGiaScaricati.has(String(voce.unitId))) return;
            risultati.push(doorNumber);
        });
        return risultati;
    }

    function confrontaBaie(a, b) {
        const aPark = /^P/i.test(a);
        const bPark = /^P/i.test(b);
        if (aPark && bPark) return parseInt(a.replace(/^P/i, ""), 10) - parseInt(b.replace(/^P/i, ""), 10);
        if (aPark) return -1;
        if (bPark) return 1;
        const na = parseInt(a, 10);
        const nb = parseInt(b, 10);
        const aNum = !isNaN(na);
        const bNum = !isNaN(nb);
        if (aNum && bNum) return na - nb;
        if (aNum) return -1;
        if (bNum) return 1;
        return a.localeCompare(b);
    }

    function renderPannelloScaricoConteggi(panelBody, gruppi) {
        let html = "";
        if (gruppi.size === 0) {
            html += '<div class="tnt-empty">Nessuno scarico in scadenza</div>';
        } else {
            const ora = new Date();
            const voci = [...gruppi.entries()].sort((a, b) => a[1].orario.getTime() - b[1].orario.getTime());
            const fasce = [
                { chiave: "ritardo", etichetta: "Cutoff scaduto", classe: "tnt-card-rosso", test: (m) => m < 0 },
                { chiave: "30", etichetta: "Cutoff entro 30 minuti", classe: "tnt-card-rosso", test: (m) => m <= 30 },
                { chiave: "60", etichetta: "Cutoff entro 60 minuti", classe: "tnt-card-ambra", test: (m) => m <= 60 },
                { chiave: "oltre", etichetta: "Prossimi cutoff", classe: "tnt-card-ambra", test: () => true }
            ];
            const bucket = new Map();
            voci.forEach(([chiave, info]) => {
                const diffMinuti = (info.orario.getTime() - ora.getTime()) / 60000;
                const fascia = fasce.find((f) => f.test(diffMinuti));
                if (!bucket.has(fascia.chiave)) bucket.set(fascia.chiave, { fascia, voci: [] });
                bucket.get(fascia.chiave).voci.push(info);
            });
            let presenteAsterisco = false;
            fasce.forEach((f) => {
                const gruppo = bucket.get(f.chiave);
                if (!gruppo || gruppo.voci.length === 0) return;
                html += '<div class="tnt-group-label">' + f.etichetta + '</div>';
                gruppo.voci.forEach((info) => {
                    const hh = String(info.orario.getHours()).padStart(2, "0");
                    const mm = String(info.orario.getMinutes()).padStart(2, "0");
                    const stessoGiornoDiOggi = info.orario.getFullYear() === ora.getFullYear() && info.orario.getMonth() === ora.getMonth() && info.orario.getDate() === ora.getDate();
                    const etichettaOrario = stessoGiornoDiOggi ? (hh + ':' + mm) : (String(info.orario.getDate()).padStart(2, "0") + "/" + String(info.orario.getMonth() + 1).padStart(2, "0") + " " + hh + ':' + mm);
                    const listaBaie = info.baie.filter(Boolean).sort(confrontaBaie).join(", ");
                    if (listaBaie.indexOf("*") !== -1) presenteAsterisco = true;
                    html += '<div class="tnt-card ' + f.classe + '">'
                        + '<div class="tnt-card-row">'
                        + '<span class="tnt-card-title">' + info.nome + ' (' + info.baie.length + ')</span>'
                        + '<span class="tnt-badge">' + etichettaOrario + '</span>'
                        + '</div>'
                        + '<div class="tnt-card-sub">Baie: ' + listaBaie + '</div>'
                        + '</div>';
                });
            });
            if (presenteAsterisco) {
                html += '<div class="tnt-footnote">* Cutoff spostato al turno successivo: arrivo troppo vicino al cutoff originale (meno di 60 minuti per i domestici, 150 per gli internazionali)</div>';
            }
        }

        const arriviAttuali = caricaArriviBaia();
        const baieLibere = elencoBaieScaricoComplete().filter((n) => {
            const voce = arriviAttuali[n];
            return !voce || !voce.unitId;
        });

        const parcheggiOccupati = elencoParcheggioScaricoOccupati().sort(confrontaBaie);
        html += '<div class="tnt-group-label">Parcheggio scarico (' + parcheggiOccupati.length + ')</div>';
        html += '<div class="tnt-card tnt-card-ambra"><div class="tnt-card-sub">'
            + (parcheggiOccupati.length > 0 ? parcheggiOccupati.join(", ") : "Nessun mezzo in parcheggio")
            + '</div></div>';

        html += '<div class="tnt-group-label">Baie libere (' + baieLibere.length + ')</div>';
        html += '<div class="tnt-card tnt-card-verde"><div class="tnt-card-sub">'
            + (baieLibere.length > 0 ? baieLibere.join(", ") : "Nessuna baia libera")
            + '</div></div>';

        panelBody.innerHTML = html;
    }

    function avviaTool() {
        impostaBadgeInAttesa();

        try { scanBlocchi(document.body); } catch (e) {}
        try { pulisciStoricoVecchio(); } catch (e) {}
        try { initIcone(); } catch (e) {}
        try { processaTripDetails(); } catch (e) {}

        let debounceTimer = null;
        const nodiDaControllare = new Set();
        function eseguiControlliPesanti() {
            const nodi = Array.from(nodiDaControllare);
            nodiDaControllare.clear();
            try { nodi.forEach((n) => { if (document.body.contains(n)) scanBlocchi(n); }); } catch (e) {}
            try { initIcone(); } catch (e) {}
            try { processaTripDetails(); } catch (e) {}
            try { aggiornaPannelloOutbound(); } catch (e) {}
        }

        function eNodoInterno(n) {
            if (!n || n.nodeType !== 1) return false;
            if (n.id && n.id.indexOf("tnt-") === 0) return true;
            return !!(n.closest && n.closest('[id^="tnt-"], .tnt-panel, .cc-menu-button'));
        }

        const observer = new MutationObserver((mutations) => {
            for (const m of mutations) {
                if (m.type === "childList") {
                    m.addedNodes.forEach((n) => { if (n.nodeType === 1 && !eNodoInterno(n)) nodiDaControllare.add(n); });
                } else if (m.type === "characterData" && m.target.parentElement) {
                    if (!eNodoInterno(m.target.parentElement)) nodiDaControllare.add(m.target.parentElement);
                } else if (m.type === "attributes" && m.target.nodeType === 1) {
                    if (!eNodoInterno(m.target)) nodiDaControllare.add(m.target);
                }
            }
            clearTimeout(debounceTimer);
            debounceTimer = setTimeout(eseguiControlliPesanti, 600);
        });

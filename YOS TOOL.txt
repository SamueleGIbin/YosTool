        if (choices.length > 1) {
            for (let i = 0; i < choices.length; i++) {
                const diffMinutiArrivoCutoff = (choices[i].cutoffDate.getTime() - arrivo.getTime()) / 6e4;
                if (diffMinutiArrivoCutoff >= margineMinimo) {
                    risultato = choices[i];
                    break;
                }
            }
            if (!risultato) risultato = choices[0];
        } else {
            risultato = choices[0];
        }
        if (arrivoPortaReale && !isNaN(arrivoPortaReale.getTime()) && arrivoPortaReale.getTime() > arrivo.getTime()) {
            const margineReale = (risultato.cutoffDate.getTime() - arrivoPortaReale.getTime()) / 6e4;
            if (margineReale < margineMinimo) {
                const cutoffOriginaleGiaPassato = risultato.cutoffDate.getTime() <= arrivoPortaReale.getTime();
                const nuovoCutoff = trovaProssimoCutoffConMargine(risultato.cat, risultato.cutoffDate, arrivoPortaReale, margineMinimo);
                if (nuovoCutoff) {
                    risultato = cutoffOriginaleGiaPassato ? {
                        cat: risultato.cat,
                        cutoffDate: nuovoCutoff,
                        ritardo: false
                    } : {
                        cat: risultato.cat,
                        cutoffDate: nuovoCutoff,
                        cutoffOriginale: risultato.cutoffDate,
                        ritardo: true,
                        margineMinimo: margineMinimo
                    };
                }
            }
        }
        if (risultato) risultato.margineMinimo = margineMinimo;
        return risultato;
    }

    function parseDataOraSlash(testo) {
        const m = testo.trim().match(/(\d{2})[\/\-](\d{2})\s+(\d{2}):(\d{2})/);
        if (!m) return null;
        const giorno = parseInt(m[1], 10), mese = parseInt(m[2], 10), ore = parseInt(m[3], 10), min = parseInt(m[4], 10);
        const ora = new Date;
        const annoCorrente = ora.getFullYear();
        let candidato = new Date(annoCorrente, mese - 1, giorno, ore, min);
        const diffGiorni = (candidato.getTime() - ora.getTime()) / 864e5;
        if (diffGiorni > 180) candidato = new Date(annoCorrente - 1, mese - 1, giorno, ore, min); else if (diffGiorni < -180) candidato = new Date(annoCorrente + 1, mese - 1, giorno, ore, min);

    function salvaArrayConPulizia(chiave, array, chiaveTempoFn) {
        let arr = array;
        let scartati = false;
        while (arr.length > 0) {
            try {
                localStorage.setItem(chiave, JSON.stringify(arr));
                return arr;
            } catch (e) {
                scartati = true;
                const giorniPresenti = new Set;
                arr.forEach(r => {
                    const t = chiaveTempoFn(r) ? new Date(chiaveTempoFn(r)).getTime() : NaN;
                    if (!isNaN(t)) giorniPresenti.add(chiaveGiorno(t));
                });
                const giornoPiuVecchio = [ ...giorniPresenti ].sort()[0];
                if (!giornoPiuVecchio) {
                    arr = [];
                    break;
                }
                arr = arr.filter(r => {
                    const t = chiaveTempoFn(r) ? new Date(chiaveTempoFn(r)).getTime() : NaN;
                    return isNaN(t) ? false : chiaveGiorno(t) !== giornoPiuVecchio;
                });
            }
        }
        try {
            localStorage.removeItem(chiave);
        } catch (e) { registraErroreSilenzioso(e); }
        return arr;
    }

    function unisciRemarks(remotoObj) {
        const limite = Date.now() - SCADENZA_REMARKS_ORE * 60 * 60 * 1e3;
        const combinate = new Map;
        Object.keys(remotoObj || {}).forEach(k => {
            const v = remotoObj[k];
            if (v && v.testo && v.ts) combinate.set(k, v);
        });
        cacheRemarksPerBaia.forEach((v, k) => {
            const esistente = combinate.get(k);
            if (!esistente || new Date(v.ts).getTime() >= new Date(esistente.ts).getTime()) combinate.set(k, v);
        });
        combinate.forEach((v, k) => {
            if (new Date(v.ts).getTime() < limite) combinate.delete(k);
        });
        return combinate;
    }
    async function sincronizzaDati() {
        if (!FIREBASE_SYNC.ATTIVO) {
            aggiornaStatoSync(false, null);
            return;
        }
        if (sincronizzazioneInCorso) {
            sincronizzazionePendente = true;
            return;
        }
        sincronizzazioneInCorso = true;
        try {
            let presenzeFinali = {};
            for (let tentativo = 0; tentativo < 6; tentativo++) {
                const remoto = await firebaseLeggiFile();
                if (remoto.dati.versione > VERSIONE_DATI) {
                    aggiornaStatoSync(false, null);
                    return;
                }
                const storicoUnito = unisciSenzaDuplicati(remoto.dati.storico, caricaStoricoReport(), chiaveStorico);
                const sessioniUnite = unisciSenzaDuplicati(remoto.dati.sessioni, caricaSessioniScarico(), chiaveSessione);
                const pulito = applicaRetentionCondivisa(storicoUnito, sessioniUnite);
                presenzeFinali = aggiornaPresenze(remoto.dati.presenze);
                const remarksUnite = unisciRemarks(remoto.dati.remarksBaie);
                cacheRemarksPerBaia.clear();
                remarksUnite.forEach((v, k) => cacheRemarksPerBaia.set(k, v));
                sessioniScaricoInCorso.forEach((v, doorNumber) => aggiornaFlagBufferSessioneInCorso(doorNumber));
                const remarksOggetto = {};
                remarksUnite.forEach((v, k) => {
                    remarksOggetto[k] = v;
                });
                const colliLocali = caricaColliLocali();
                const colliRemoti = remoto.dati.colliTotali;
                let colliFinali = colliRemoti;
                if (colliLocali && (!colliRemoti || new Date(colliLocali.aggiornatoIl).getTime() > new Date(colliRemoti.aggiornatoIl).getTime())) {
                    colliFinali = colliLocali;
                }
                if (colliFinali) salvaColliLocali(colliFinali);
                aggiornaVisualizzazioneColli();
                const colliStoricoUnito = unisciColliStorico(remoto.dati.colliStorico, caricaColliStorico());
                salvaColliStorico(colliStoricoUnito);
                salvaStoricoReport(pulito.storico);
                salvaSessioniScarico(pulito.sessioni);
                const invariato = pulito.storico.length === remoto.dati.storico.length && pulito.sessioni.length === remoto.dati.sessioni.length && remoto.dati.presenze[DEVICE_ID] === presenzeFinali[DEVICE_ID] && Object.keys(presenzeFinali).length === Object.keys(remoto.dati.presenze).length && JSON.stringify(remarksOggetto) === JSON.stringify(remoto.dati.remarksBaie || {}) && JSON.stringify(colliFinali) === JSON.stringify(colliRemoti) && colliStoricoUnito.length === remoto.dati.colliStorico.length;
                if (invariato) break;
                const ok = await firebaseScriviFile({
                    versione: VERSIONE_DATI,
                    storico: pulito.storico,
                    sessioni: pulito.sessioni,
                    presenze: presenzeFinali,
                    remarksBaie: remarksOggetto,
                    colliTotali: colliFinali || null,
                    colliStorico: colliStoricoUnito
                }, remoto.etag);
                if (ok) break;
                await attesa(400 * (tentativo + 1) + Math.random() * 300);
            }
            aggiornaStatoSync(true, contaUtentiOggi(presenzeFinali));
        } catch (e) {
            const motivo = /\s(401|403)\b/.test(e.message || "") ? "token non valido" : null;
            aggiornaStatoSync(false, null, motivo);
        } finally {
            sincronizzazioneInCorso = false;
            if (sincronizzazionePendente) {
                sincronizzazionePendente = false;
                pianificaSincronizzazione(2e3);
            }
        }
    }

    function costruisciDataOra(dataStr, oraStr, oraFallback) {
        if (!dataStr) return null;
        const parti = dataStr.split("-").map(Number);
        const anno = parti[0], mese = parti[1], giorno = parti[2];
        const oraEffettiva = oraStr || oraFallback || "00:00";
        const partiOra = oraEffettiva.split(":").map(Number);
        const ore = partiOra[0] || 0, minuti = partiOra[1] || 0;
        return new Date(anno, mese - 1, giorno, ore, minuti, 0, 0);
    }
    function filtraPerPeriodoConOra(lista, dataInizioStr, oraInizioStr, dataFineStr, oraFineStr, campoData) {
        campoData = campoData || "timestampIso";
        if (!dataInizioStr && !dataFineStr) return lista;
        const inizio = dataInizioStr ? costruisciDataOra(dataInizioStr, oraInizioStr, "00:00") : null;
        const fine = dataFineStr ? costruisciDataOra(dataFineStr, oraFineStr, "23:59") : null;
        return lista.filter(r => {
            const iso = r[campoData];
            if (!iso) return true;
            const d = new Date(iso);
            if (isNaN(d.getTime())) return true;
            if (inizio && d < inizio) return false;
            if (fine && d > fine) return false;
            return true;
        });
    }
    function filtraPerPeriodo(lista, dataInizioStr, dataFineStr, campoData) {
        campoData = campoData || "timestampIso";
        if (!dataInizioStr && !dataFineStr) return lista;
        const inizio = dataInizioStr ? new Date(dataInizioStr + "T00:00:00") : null;
        const fine = dataFineStr ? new Date(dataFineStr + "T23:59:59") : null;
        return lista.filter(r => {
            const iso = r[campoData];
            if (!iso) return true;
            const d = new Date(iso);
            if (isNaN(d.getTime())) return true;
            if (inizio && d < inizio) return false;
            if (fine && d > fine) return false;
            return true;
        });
    }

    function calcolaAttesaMediaParcheggio(storico) {
        const perUnit = new Map;
        storico.forEach(r => {
            if (!r.unitId || r.unitId === "N/A") return;
            const cat = categoriaBaia(r.doorNumber);
            if (cat !== "Parcheggio" && cat !== "Scarico") return;
            const t = r.timestampIso ? new Date(r.timestampIso).getTime() : NaN;
            if (isNaN(t)) return;
            if (!perUnit.has(r.unitId)) perUnit.set(r.unitId, []);
            perUnit.get(r.unitId).push({
                cat: cat,
                t: t
            });
        });
        const attese = [];
        perUnit.forEach(eventi => {
            eventi.sort((a, b) => a.t - b.t);
            const primoParcheggio = eventi.find(e => e.cat === "Parcheggio");
            if (!primoParcheggio) return;
            const primaBaiaDopo = eventi.find(e => e.cat === "Scarico" && e.t > primoParcheggio.t);
            if (!primaBaiaDopo) return;
            const minuti = (primaBaiaDopo.t - primoParcheggio.t) / 6e4;
            if (minuti >= 0 && minuti < 24 * 60) attese.push(minuti);
        });
        if (attese.length === 0) { return {
            mediaMinuti: null,
            campioni: 0
        }; }
        const media = attese.reduce((s, v) => s + v, 0) / attese.length;
        return {
            mediaMinuti: Math.round(media),
            campioni: attese.length
        };
    }
    function calcolaTempoScaricoPerLato(dataInizioStr, oraInizioStr, dataFineStr, oraFineStr) {
        const sessioni = filtraPerPeriodoConOra(caricaSessioniScarico(), dataInizioStr, oraInizioStr, dataFineStr, oraFineStr, "inizioIso");
        const ordineLati = [ "Lato 100", "Lato Furgoni", "Lato 200", "Lato 600", "Lato 500" ];
        const gruppi = new Map;
        ordineLati.forEach(l => gruppi.set(l, []));
        sessioni.forEach(s => {
            if (s.buffer) return;
            if (!s.lato || typeof s.durataMinuti !== "number" || isNaN(s.durataMinuti)) return;
            if (!gruppi.has(s.lato)) gruppi.set(s.lato, []);
            gruppi.get(s.lato).push(s.durataMinuti);
        });
        const righe = [];
        ordineLati.forEach(lato => {
            const valori = gruppi.get(lato) || [];
            if (valori.length === 0) {
                righe.push({
                    chiave: lato,
                    mediaMinuti: null,
                    campioni: 0
                });
                return;
            }
            const media = valori.reduce((s, v) => s + v, 0) / valori.length;
            righe.push({
                chiave: lato,
                mediaMinuti: Math.round(media),
                campioni: valori.length
            });
        });
        return righe;
    }
    function calcolaTempoScaricoPerBaia(dataInizioStr, oraInizioStr, dataFineStr, oraFineStr) {
        const sessioni = filtraPerPeriodoConOra(caricaSessioniScarico(), dataInizioStr, oraInizioStr, dataFineStr, oraFineStr, "inizioIso");
        const gruppi = new Map;
        sessioni.forEach(s => {
            if (s.buffer) return;
            if (!s.doorNumber || typeof s.durataMinuti !== "number" || isNaN(s.durataMinuti)) return;
            if (!gruppi.has(s.doorNumber)) gruppi.set(s.doorNumber, []);
            gruppi.get(s.doorNumber).push(s.durataMinuti);
        });
        const medieLato = new Map;
        calcolaTempoScaricoPerLato(dataInizioStr, oraInizioStr, dataFineStr, oraFineStr).forEach(r => medieLato.set(r.chiave, r.mediaMinuti));
        const righe = [];
        gruppi.forEach((valori, doorNumber) => {
            const media = valori.reduce((s, v) => s + v, 0) / valori.length;
            const lato = latoBaia(doorNumber);
            const mediaLato = lato ? medieLato.get(lato) : null;
            const scartoVsLato = mediaLato !== null && mediaLato !== undefined ? Math.round(media - mediaLato) : null;
            righe.push({
                doorNumber: doorNumber,
                lato: lato || "",
                mediaMinuti: Math.round(media),
                campioni: valori.length,
                scartoVsLato: scartoVsLato
            });
        });
        righe.sort((a, b) => b.mediaMinuti - a.mediaMinuti);
        return righe;
    }
    function calcolaFinestraTrend(dataInizioStr) {
        if (!dataInizioStr) return null;
        const inizioPeriodo = new Date(dataInizioStr + "T00:00:00");
        if (isNaN(inizioPeriodo.getTime())) return null;
        const fineTrend = new Date(inizioPeriodo.getTime() - 24 * 60 * 60 * 1e3);
        const inizioTrend = new Date(fineTrend.getTime() - (GIORNI_MEDIA_TREND - 1) * 24 * 60 * 60 * 1e3);
        return {
            dataInizioStr: inizioTrend.toISOString().slice(0, 10),
            dataFineStr: fineTrend.toISOString().slice(0, 10)
        };
    }
    function calcolaTrend(dataInizioStr) {
        const finestra = calcolaFinestraTrend(dataInizioStr);
        if (!finestra) return null;
        const perLato = calcolaTempoScaricoPerLato(finestra.dataInizioStr, "00:00", finestra.dataFineStr, "23:59");
        const mappaLato = new Map;
        perLato.forEach(r => mappaLato.set(r.chiave, r.mediaMinuti));
        const storicoFinestra = filtraPerPeriodoConOra(caricaStoricoReport(), finestra.dataInizioStr, "00:00", finestra.dataFineStr, "23:59", "timestampIso");
        const attesaParcheggio = calcolaAttesaMediaParcheggio(storicoFinestra);
        return {
            perLato: mappaLato,
            attesaParcheggioMediaMinuti: attesaParcheggio.mediaMinuti,
            giorni: GIORNI_MEDIA_TREND
        };
    }
    function calcolaRitardiPer(storico, chiaveFn, ordinaPer) {
        const gruppi = new Map;
        storico.forEach(r => {
            const chiave = chiaveFn(r);
            if (!chiave || chiave === "N/A" || chiave === "N/D") return;
            if (!gruppi.has(chiave)) { gruppi.set(chiave, {
                totale: 0,
                ritardati: 0
            }); }
            const g = gruppi.get(chiave);
            g.totale++;
            if (r.ritardoFlag === "Si") g.ritardati++;
        });
        const righe = [];
        gruppi.forEach((v, chiave) => {
            righe.push({
                chiave: chiave,
                totale: v.totale,
                ritardati: v.ritardati,
                percentuale: v.totale > 0 ? Math.round(v.ritardati / v.totale * 100) : 0
            });
        });
        if (ordinaPer === "percentuale") righe.sort((a, b) => b.percentuale - a.percentuale); else righe.sort((a, b) => b.totale - a.totale);
        return righe;
    }
    function calcolaStatisticheTurno(dataInizioStr, oraInizioStr, dataFineStr, oraFineStr) {

            return cat === "Scarico" || cat === "Parcheggio";
        });
        const camionPresiInCarico = storico.filter(r => categoriaBaia(r.doorNumber) === "Scarico").length;
        return {
            totaleRecord: storicoScaricoParcheggio.length,
            camionPresiInCarico: camionPresiInCarico,
            attesaParcheggio: calcolaAttesaMediaParcheggio(storico),
            perLato: calcolaTempoScaricoPerLato(dataInizioStr, oraInizioStr, dataFineStr, oraFineStr),
            perBaia: calcolaTempoScaricoPerBaia(dataInizioStr, oraInizioStr, dataFineStr, oraFineStr),

            perCliente: calcolaRitardiPer(storicoScaricoParcheggio, r => r.codiceOrigine, "percentuale"),
            trend: calcolaTrend(dataInizioStr)

    function finestraTurnoCorrente(riferimento) {
        const ora = riferimento || new Date;
        const minutiOra = ora.getHours() * 60 + ora.getMinutes();
        const y = ora.getFullYear(), mo = ora.getMonth(), d = ora.getDate();
        function dataOra(offsetGiorni, h, m) {
            return new Date(y, mo, d + offsetGiorni, h, m, 0, 0);
        }
        if (minutiOra >= 21 * 60 + 30) { return {
            inizio: dataOra(0, 21, 30),
            fine: dataOra(1, 1, 0),
            nome: "Pomeriggio"
        }; }
        if (minutiOra < 60) { return {
            inizio: dataOra(-1, 21, 30),
            fine: dataOra(0, 1, 0),
            nome: "Pomeriggio"
        }; }
        if (minutiOra < 9 * 60) { return {
            inizio: dataOra(0, 1, 0),
            fine: dataOra(0, 9, 0),
            nome: "Notte"
        }; }
        return {
            inizio: dataOra(0, 21, 30),
            fine: dataOra(1, 1, 0),
            nome: "Pomeriggio"
        };
    }
    function leggiCutoffBaiaCorrente() {
        const originClassEl = document.querySelector(".inbound-origin-en-route, .inbound-origin");
        let codiceOrigine = null;
        if (originClassEl && originClassEl.textContent.trim()) codiceOrigine = estraiCodiceGrezzo(originClassEl);
        const contenitori = document.querySelectorAll(".door-info__movmt-no-left-padding");
        for (const contenitore of contenitori) {
            const sezione = contenitore.querySelector(".door-info__movmt-section");
            if (!sezione) continue;
            if (!codiceOrigine) codiceOrigine = estraiCodiceGrezzo(trovaValorePerChiave(sezione, "Origin"));
            const arrivoEl = trovaValorePerChiave(sezione, "Sched. Arrival");
            if (!arrivoEl || !codiceOrigine) continue;
            const testoArrivo = arrivoEl.textContent.trim();
            if (testoArrivo.toUpperCase().includes("N/A")) continue;
            let arrivo = parseDataOraSlash(testoArrivo);
            if (!arrivo) continue;
            const actualArrivoEl = trovaValorePerChiave(sezione, "Actual Arrival");
            if (actualArrivoEl) {
                const testoActual = actualArrivoEl.textContent.trim();
                if (testoActual && !testoActual.toUpperCase().includes("N/A")) {
                    const arrivoReale = parseDataOraSlash(testoActual);
                    if (arrivoReale) arrivo = arrivoReale;
                }
            }
            let arrivoPortaReale = null;
            if (baiaSelezionata) {
                const arriviSalvati = caricaArriviBaia();
                const voceBaia = arriviSalvati[baiaSelezionata];
                if (voceBaia && voceBaia.orario) {
                    const d = new Date(voceBaia.orario);
                    if (!isNaN(d.getTime())) arrivoPortaReale = d;
                }
            }
            const cutoffRes = calcolaCutoffPerCodice(codiceOrigine, arrivo, arrivoPortaReale);
            if (cutoffRes) { return {
                cutoff: cutoffRes,
                codice: codiceOrigine
            }; }
        }
        return null;
    }

    function scansionaBaieScarico(panelBody) {
        const gruppi = new Map;
        const ora = new Date;
        const finestra = finestraTurnoCorrente(ora);
        function aggiungiAlGruppo(risultato, infoBaia) {
            if (risultato.cutoffDate.getTime() < finestra.inizio.getTime() || risultato.cutoffDate.getTime() > finestra.fine.getTime()) return;
            const senzaBaiaAssegnata = /^\(.*\)$/.test(infoBaia);
            if (senzaBaiaAssegnata) return;
            const hh = String(risultato.cutoffDate.getHours()).padStart(2, "0");
            const mm = String(risultato.cutoffDate.getMinutes()).padStart(2, "0");
            const dataChiave = risultato.cutoffDate.getFullYear() + "-" + String(risultato.cutoffDate.getMonth() + 1).padStart(2, "0") + "-" + String(risultato.cutoffDate.getDate()).padStart(2, "0");
            const chiave = risultato.cat.nome + "|" + dataChiave + " " + hh + ":" + mm;
            if (!gruppi.has(chiave)) { gruppi.set(chiave, {
                nome: risultato.cat.nome,
                turno: risultato.cat.turno,
                orario: risultato.cutoffDate,
                baie: []
            }); }
            gruppi.get(chiave).baie.push(risultato.ritardo ? infoBaia + "*" : infoBaia);
        }
        const baie = trovaBaieScaricoOccupate();
        const arriviCorrenti = caricaArriviBaia();
        for (const baia of baie) {
            if (baiaBloccataScarico(baia.doorNumber)) continue;
            const statoBordo = statoBordoBaiaScarico(baia.doorNumber);
            if (statoBordo === "pronta" || statoBordo === "non_attiva" || statoBordo === "shunter") continue;
            const voceBaia = arriviCorrenti[baia.doorNumber];
            if (categoriaBaia(baia.doorNumber) === "Parcheggio" && voceBaia && unitGiaScaricati.has(String(voceBaia.unitId))) continue;
            let infoCutoff = calcolaCutoffDaCacheMovimenti(baia.doorNumber);
            if (!infoCutoff) {
                const cutoffCachato = leggiCacheCutoffBaia(baia.doorNumber);
                if (cutoffCachato) { infoCutoff = {
                    cutoff: cutoffCachato,
                    codice: null
                }; }
            }
            if (!infoCutoff && baiaSelezionata === baia.doorNumber) {
                assicuraTabInbound();
                infoCutoff = leggiCutoffBaiaCorrente();
                if (infoCutoff) impostaCacheCutoffBaia(baia.doorNumber, infoCutoff.cutoff, infoCutoff.codice);
            }
            if (!infoCutoff) continue;
            aggiungiAlGruppo(infoCutoff.cutoff, baia.doorNumber);
        }
        renderPannelloScaricoConteggi(panelBody, gruppi, finestra);
    }

    function elencoParcheggioScaricoOccupati() {
        const arrivi = caricaArriviBaia();

    function confrontaBaie(a, b) {
        const aPark = /^P/i.test(a);
        const bPark = /^P/i.test(b);
        if (aPark && bPark) return parseInt(a.replace(/^P/i, ""), 10) - parseInt(b.replace(/^P/i, ""), 10);
        if (aPark) return -1;
        if (bPark) return 1;
        const na = parseInt(a, 10);
        const nb = parseInt(b, 10);
        const aNum = !isNaN(na);
        const bNum = !isNaN(nb);
        if (aNum && bNum) return na - nb;
        if (aNum) return -1;
        if (bNum) return 1;
        return a.localeCompare(b);
    }

        function eNodoInterno(n) {
            if (!n || n.nodeType !== 1) return false;
            if (n.id && n.id.indexOf("tnt-") === 0) return true;
            return !!(n.closest && n.closest('[id^="tnt-"], .tnt-panel, .cc-menu-button'));
        }
        const observer = new MutationObserver(mutations => {
            for (const m of mutations) {
                if (m.type === "childList") {
                    m.addedNodes.forEach(n => {
                        if (n.nodeType === 1 && !eNodoInterno(n)) nodiDaControllare.add(n);
                    });
                } else if (m.type === "characterData" && m.target.parentElement) {
                    if (!eNodoInterno(m.target.parentElement)) nodiDaControllare.add(m.target.parentElement);
                } else if (m.type === "attributes" && m.target.nodeType === 1) {
                    if (!eNodoInterno(m.target)) nodiDaControllare.add(m.target);
                }
            }
            clearTimeout(debounceTimer);
            debounceTimer = setTimeout(eseguiControlliPesanti, 600);
        });
